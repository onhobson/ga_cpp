29 constraints::HasValueType<Sequence> &&
30 constraints::Hashable<typename Sequence::value_type>)
31 Sequence alternatingEdge(const Sequence& parent1, const Sequence& parent2, Random& rng)
32 {
33 const std::size_t size { std::ranges::size(parent1) };
34 if (size == 0 || size != std::ranges::size(parent2))
35 throw std::invalid_argument("Invalid sequence size");
36
37 using Gene = typename Sequence::value_type;
38
39 const auto parent1Begin { std::ranges::begin(parent1) };
40 std::unordered_set<Gene, ga::Hash<Gene>> used {};
41 Sequence child { parent1 };
42 auto childBegin { std::ranges::begin(child) };
43
44 const std::size_t startIndex { rng.template getInteger<std::size_t>(0, size - 1) };
45 Gene current { parent1Begin[startIndex] };
46 std::size_t childSize {};
47 childBegin[childSize++] = current;
48 used.insert(current);
49
50 bool parentOne { (rng.getInteger(0, 1)) ? true : false };
51
52 while (childSize < size)
53 {
54 // Switch the parent used every iteration
55 const Sequence& parent { parentOne ? parent1 : parent2 };
56 const auto parentBegin { std::ranges::begin(parent) };
57 const auto parentEnd { std::ranges::end(parent) };
58
59 const auto currentPosition { std::ranges::find(parent, current) };
60 std::vector<Gene> candidates {};
61
62 if (currentPosition != parentEnd)
63 {
64 // Get next Gene after current
65 const std::size_t index { static_cast<std::size_t>(currentPosition - parentBegin) };
66 const Gene& next { parentBegin[(index + 1) % size] };
67
68 if (!used.contains(next))
69 {
70 candidates.push_back(next);
71 }
72 }
73
74 parentOne = !parentOne;
75
76 if (candidates.empty())
77 {
78 // Add every Gene not yet used
79 for (const Gene& gene : parent1)
80 {
81 if (!used.contains(gene))
82 {
83 candidates.push_back(gene);
84 }
85 }
86 }
87
88 const std::size_t choice { rng.template getInteger<std::size_t>(0, candidates.size() - 1) };
89 current = candidates[choice];
90 childBegin[childSize++] = current;
91 used.insert(current);
92 }
93
94 return child;
95 }

108 template <typename Sequence>
109 requires(constraints::Permutable<Sequence> &&
110 constraints::HasValueType<Sequence> &&
111 constraints::Hashable<typename Sequence::value_type>)
112 Sequence order(const Sequence& parent1, const Sequence& parent2, Random& rng)
113 {
114 const std::size_t size { std::ranges::size(parent1) };
115 if (size < 2 || size != std::ranges::size(parent2))
116 throw std::invalid_argument("Invalid sequence size");
117
118 using Gene = typename Sequence::value_type;
119
120 Sequence child { parent1 };
121 std::unordered_set<Gene, ga::Hash<Gene>> used {};
122
123 const auto parent1Begin { std::ranges::begin(parent1) };
124 const auto parent2Begin { std::ranges::begin(parent2) };
125 auto childBegin { std::ranges::begin(child) };
126
127 // Copy subsequence from parent1
128 const std::size_t firstCut { rng.template getInteger<std::size_t>(0, size - 1) };
129 const std::size_t secondCut { rng.template getInteger<std::size_t>(firstCut + 1, size) };
130
131 for (std::size_t i { firstCut }; i < secondCut; ++i)
132 used.insert(parent1Begin[i]);
133
134 // Copy remaining from parent2 using vertex ordering
135 std::size_t childIndex { (secondCut < size) ? secondCut : 0 };
136 for (std::size_t offset { 0 }; offset < size; ++offset)
137 {
138 const Gene& gene { parent2Begin[(secondCut + offset) % size] };
139
140 if (used.contains(gene))
141 continue;
142
143 childBegin[childIndex] = gene;
144 used.insert(gene);
145 childIndex = (childIndex + 1) % size;
146 }
147
148 return child;
149 }

161 template <typename Sequence>
162 requires(constraints::Permutable<Sequence> &&
163 constraints::HasValueType<Sequence> &&
164 constraints::Hashable<typename Sequence::value_type>)
165 Sequence edgeRecombination(const Sequence& parent1, const Sequence& parent2, Random& rng)
166 {
167 const std::size_t size { std::ranges::size(parent1) };
168 if (size == 0 || size != std::ranges::size(parent2))
169 throw std::invalid_argument("Invalid sequence size");
170
171 using Gene = typename Sequence::value_type;
172 const auto parent1Begin { std::ranges::begin(parent1) };
173
174 // Create adjacency lists
175 using AdjList = std::unordered_set<Gene, ga::Hash<Gene>>;
176
177 std::unordered_map<Gene, AdjList, ga::Hash<Gene>> adjPar1;
178 std::unordered_map<Gene, AdjList, ga::Hash<Gene>> adjPar2;
179 std::unordered_map<Gene, AdjList, ga::Hash<Gene>> adjCombined;
180
181 for (int parentChoice : { 0, 1 })
182 {
183 const Sequence& parent { (parentChoice) ? parent2 : parent1 };
184 auto& adjList { (parentChoice) ? adjPar2 : adjPar1 };
185
186 const auto parentBegin { std::ranges::begin(parent) };
187
188 for (std::size_t i { 0 }; i < size; ++i)
189 {
190 // Add values to the left and right to adj list
191 const std::size_t left { (i + size - 1) % size };
192 const std::size_t right { (i + 1) % size };
193
194 if (adjList.contains(parentBegin[i]))
195 throw std::invalid_argument("seq" + std::to_string(parentChoice + 1) + " is not a permutation");
196
197 adjList[parentBegin[i]].insert({ parentBegin[left], parentBegin[right] });
198 }
199 }
200
201 // Verify genes of both sequences match
202 for (const auto& [gene, _] : adjPar1)
203 {
204 if (!adjPar2.contains(gene))
205 throw std::invalid_argument("Parent sequences must be permutations of the same gene sets");
206 }
207
208 // Combine adj lists
209 for (const auto& [gene, adjList] : adjPar1)
210 {
211 adjCombined[gene].insert(adjList.begin(), adjList.end());
212 adjCombined[gene].insert(adjPar2[gene].begin(), adjPar2[gene].end());
213 }
214
215 Sequence child { parent1 };
216 auto childBegin { std::ranges::begin(child) };
217
218 Gene gene { parent1Begin[0] };
219 std::unordered_set<Gene, ga::Hash<Gene>> used;
220 std::size_t childSize {};
221 childBegin[childSize++] = gene;
222 used.insert(gene);
223
224 while (childSize < size)
225 {
226 // Remove gene from all adj lists
227 for (auto& [_, adjList] : adjCombined)
228 {
229 const auto it { adjList.find(gene) };
230
231 if (it != adjList.end())
232 adjList.erase(it);
233 }
234
235 if (!adjCombined[gene].empty())
236 {
237 // Set next to gene neighbor with fewest neighbors
238 std::vector<Gene> candidates;
239 std::size_t minNeighborSize = std::numeric_limits<std::size_t>::max();
240
241 for (const Gene& neighbor : adjCombined[gene])
242 {
243 std::size_t neighborSize { adjCombined[neighbor].size() };
244
245 if (neighborSize < minNeighborSize)
246 {
247 candidates.clear();
248 candidates.push_back(neighbor);
249 minNeighborSize = neighborSize;
250 }
251 else if (neighborSize == minNeighborSize)
252 {
253 candidates.push_back(neighbor);
254 }
255 }
256
257 // Select random candidate
258 gene = candidates[ rng.getInteger<std::size_t>(0, candidates.size() - 1) ];
259 }
260 else
261 {
262 // Set next to random gene not yet used
263 std::vector<Gene> candidates;
264
265 for (const auto& candidate : parent1)
266 {
267 if (!used.contains(candidate))
268 candidates.push_back(candidate);
269 }
270
271 gene = candidates[ rng.getInteger<std::size_t>(0, candidates.size() - 1) ];
272 }
273
274 childBegin[childSize++] = gene;
275 used.insert(gene);
276 }
277
278 return child;
279 }
