132 template <typename Initializer, typename Terminator, typename Callback>
133 requires (
134 constraints::TerminationCondition<Terminator, Fitness> &&
135 constraints::CallbackFunction<Callback, Sequence, Fitness>
136 )
137 Result<Sequence, Fitness> run(
138 const std::vector<Sequence>& population,
139 RunOptions<Initializer, Terminator, Callback> options
140 ) {
141 if (population.size() != m_config.populationSize)
142 throw std::invalid_argument(std::format(
143 "Given population size was not equal to specified population size in Config object. {} vs {}", population.size(), m_config.populationSize
144 ));
145
146 // Default all tracking members
147 m_bestSolution = {};
148 m_sinceLastCallback = 0;
149 m_generationCount = 0;
150
151 initializePopulation(population);
152 evaluateCurrentGeneration();
153 m_bestSolution = m_currentPopulation.front();
154
155 // Begin GA loop
156 while (!options.terminator(createTerminationContext()))
157 {
158 ++m_generationCount;
159
160 createNewGeneration();
161
162 evaluateCurrentGeneration();
163
164 // const ref to ensure callback does not modify members
165 const std::vector<Chromosome>& currentPopulation = m_currentPopulation;
166 const Chromosome& bestSolution = m_bestSolution;
167
168 // Call callback if interval not set to 0 and interval reached
169 if (m_config.callbackInterval && ++m_sinceLastCallback >= m_config.callbackInterval)
170 {
171 std::invoke(options.callback, createCallbackContext());
172 m_sinceLastCallback = 0;
173 }
174 }
175
176 return Result {
177 .best = m_bestSolution,
178 .generation = m_generationCount
179 };
180 }

198 // Constructs Chromosomes out of each member in population with default Fitness
199 void initializePopulation(const std::vector<Sequence>& population)
200 {
201 m_currentPopulation.clear();
202 m_currentPopulation.reserve(m_config.populationSize);
203
204 for (const auto& sequence : population)
205 m_currentPopulation.emplace_back( Chromosome { .sequence = sequence } );
206 }
207
208 // Creates a new generation of chromosomes by preserving elites and generating offspring from the current population.
209 void createNewGeneration()
210 {
211 std::vector<Chromosome> newGeneration {};
212 newGeneration.reserve(m_config.populationSize);
213
214 // Carry over elites
215 for (std::size_t i { 0 }; i < m_config.eliteCount; ++i)
216 newGeneration.push_back(m_currentPopulation[i]);
217
218 // Fill rest of population
219 while (newGeneration.size() < m_config.populationSize)
220 {
221 std::size_t firstIndex { getRandomPopulationIndex() };
222 std::size_t secondIndex { getRandomPopulationIndex(firstIndex) };
223
224 const auto [first, second] { m_selector.select(m_currentPopulation) };
225
226 newGeneration.push_back(
227 Chromosome {
228 m_populator.createSequence(
229 first.sequence,
230 second.sequence,
231 m_rng
232 ), 0
233 }
234 );
235 }
236
237 m_currentPopulation = std::move(newGeneration);
238 }
239
240 // Evaluates the fitness of the current generation and updates the elite min-heap accordingly.
241 void evaluateCurrentGeneration()
242 {
243 m_evaluator.evaluate(m_currentPopulation);
244
245 if (m_compare(m_currentPopulation.front().fitness, m_bestSolution.fitness))
246 m_bestSolution = m_currentPopulation.front();
247 }
