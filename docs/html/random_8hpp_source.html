89 template <std::integral T = int>
90 T getIntegerExclude(T min, T max, const std::vector<T>& exclude) {
91 if (min > max)
92 throw std::invalid_argument("min value greater than max value");
93
94 // Clean exclude: bound, sort & remove duplicates
95 std::vector<T> excludedInRange;
96 for (const T value : exclude) {
97 if (value >= min && value <= max)
98 excludedInRange.push_back(value);
99 }
100 std::ranges::sort(excludedInRange);
101 excludedInRange.erase(std::unique(excludedInRange.begin(), excludedInRange.end()), excludedInRange.end());
102
103 // Iterate each possible value until one is not contained in exclude list
104 T expected = min;
105 bool allExcluded = true;
106 bool reachedMax = false;
107 for (const T value : excludedInRange) {
108 if (value != expected) {
109 allExcluded = false;
110 break;
111 }
112 if (value == max) {
113 reachedMax = true;
114 break;
115 }
116 ++expected;
117 }
118 if (allExcluded && reachedMax)
119 throw std::invalid_argument("Impossible request: every value in range is excluded");
120
121 std::uniform_int_distribution<T> dist {min, max};
122 T num;
123 do {
124 num = dist(m_impl);
125 } while (std::ranges::binary_search(excludedInRange, num));
126
127 return num;
128 }
